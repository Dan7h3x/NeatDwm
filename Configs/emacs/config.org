  (package-initialize)

            '(("gnu"    . "https://elpa.gnu.org/packages/")

*** Detect and activate (per buffer)
Every Python buffer activates the environment of /its/ project, buffer-locally
(=VIRTUAL_ENV=, =PATH=, =exec-path=, =run-python=), so projects with different
venvs coexist. Order of precedence: your fzf choice for the project (remembered
across restarts), then the nearest =.venv= / =venv= above the file. It runs before
=eglot-ensure=, so =ty= starts inside the right environment and =ruff= comes from
the venv when installed there.
#+begin_src emacs-lisp
(defvar my/venv-names '(".venv" "PyEnv"))
(defvar my/venv-search-dirs '("~/PyEnv/" "~/.local/share/virtualenvs" "~/.conda/envs")
  "Extra directories whose children are offered as environments in the picker.")
(defvar my/venv-choices nil
  "Alist (PROJECT-ROOT . VENV-DIR or \"none\"); persisted by savehist.")

(defun my/venv-find (&optional dir)
  "Nearest virtualenv directory above DIR (default: the buffer's directory)."
  (let ((dir (or dir default-directory)))
    (cl-some (lambda (name)
               (when-let* ((root (locate-dominating-file dir name))
                           (venv (expand-file-name name root)))
                 (and (file-executable-p (expand-file-name "bin/python" venv)) venv)))
             my/venv-names)))

(defun my/venv--current ()
  "Environment for this buffer's project: remembered choice, else detected."
  (let ((choice (alist-get (my/project-root) my/venv-choices nil nil #'equal)))
    (cond ((equal choice "none") nil)
          ((and choice (file-executable-p (expand-file-name "bin/python" choice))) choice)
          (t (my/venv-find)))))

(defun my/venv--label (venv)
  (concat (file-name-nondirectory (directory-file-name (file-name-directory venv)))
          "/" (file-name-nondirectory venv)))

(defun my/venv-activate ()
  "Buffer-locally activate this project's virtualenv (or deactivate)."
  (if-let* ((venv (my/venv--current)))
      (let ((bin  (expand-file-name "bin" venv))
            (path (getenv-internal "PATH" (default-value 'process-environment))))
        (setq-local python-shell-virtualenv-root venv)
        (setq-local exec-path (cons bin (remove bin (default-value 'exec-path))))
        (setq-local process-environment
                    (append (list (concat "VIRTUAL_ENV=" venv)
                                  (concat "PATH=" bin ":" path))
                            (cl-remove-if
                             (lambda (e) (string-match-p "\\`\\(VIRTUAL_ENV\\|PATH\\)=" e))
                             (default-value 'process-environment))))
        (setq-local mode-line-process (format " [%s]" (my/venv--label venv))))
    (dolist (v '(python-shell-virtualenv-root exec-path process-environment mode-line-process))
      (kill-local-variable v))))

;; depth -90: run before `eglot-ensure' so the server inherits the env
(add-hook 'python-base-mode-hook #'my/venv-activate -90)
#+end_src

*** Pick with fzf (existing venvs, uv interpreters, uv sync, none)
=SPC c v= (or =SPC m v= in Python buffers). Choices: venvs found for this project
and in =my/venv-search-dirs=; =uv sync= when a =pyproject.toml= exists; "new
=.venv=" with any interpreter =uv= knows about (=uv python list=); or none. The
selection is remembered per project, applied to all its Python buffers, and the
language server restarts inside the new environment.
#+begin_src emacs-lisp
(defun my/venv--version (venv)
  "Python version from VENV's pyvenv.cfg, or nil."
  (let ((cfg (expand-file-name "pyvenv.cfg" venv)))
    (when (file-readable-p cfg)
      (with-temp-buffer
        (insert-file-contents cfg)
        (when (re-search-forward "^version\\(?:_info\\)?[ \t]*=[ \t]*\\([0-9.]+\\)" nil t)
          (match-string 1))))))

(defun my/venv--existing (root)
  "Existing virtualenv directories relevant to ROOT, without duplicates."
  (let (found)
    (dolist (name my/venv-names)
      (when-let* ((r (locate-dominating-file root name)))
        (push (expand-file-name name r) found)))
    (dolist (d my/venv-search-dirs)
      (let ((d (expand-file-name d)))
        (when (file-directory-p d)
          (dolist (v (directory-files d t "\\`[^.]" t))
            (push v found)))))
    (seq-uniq
     (seq-filter (lambda (v) (file-executable-p (expand-file-name "bin/python" v)))
                 (nreverse found))
     #'equal)))

(defun my/uv--pythons ()
  "Interpreters uv knows: list of (PATH VERSION IMPLEMENTATION), deduplicated."
  (when (executable-find "uv")
    (condition-case nil
        (with-temp-buffer
          (when (zerop (call-process "uv" nil '(t nil) nil
                                     "python" "list" "--only-installed"
                                     "--output-format" "json"))
            (goto-char (point-min))
            (let (seen out)
              (dolist (o (json-parse-buffer :array-type 'list))
                (let* ((path (gethash "path" o))
                       (true (and (stringp path) (file-truename path))))
                  (when (and true (not (member true seen)))
                    (push true seen)
                    (push (list true (gethash "version" o) (gethash "implementation" o)) out))))
              (nreverse out))))
      (error nil))))

(defun my/venv--candidates (root)
  "Alist (DISPLAY-LINE . ACTION) for the picker."
  (let ((cur (my/venv--current)) out)
    (dolist (v (my/venv--existing root))
      (push (cons (format "venv  %s%s%s"
                          (abbreviate-file-name v)
                          (if-let* ((ver (my/venv--version v))) (format "  (%s)" ver) "")
                          (if (equal v cur) "  *" ""))
                  (list 'use v))
            out))
    (when (and (executable-find "uv")
               (file-exists-p (expand-file-name "pyproject.toml" root)))
      (push (cons "uv    sync  (create/refresh .venv from pyproject.toml)" '(sync)) out))
    (dolist (p (my/uv--pythons))
      (push (cons (format "uv    new   .venv with %s %s  (%s)"
                          (nth 2 p) (nth 1 p) (abbreviate-file-name (car p)))
                  (list 'new (car p)))
            out))
    (push (cons "none  deactivate (use the global python)" '(none)) out)
    (nreverse out)))

(defun my/uv--run (root args on-success)
  "Run `uv ARGS' in ROOT asynchronously (output in *uv*); call ON-SUCCESS if it works."
  (unless (executable-find "uv") (user-error "uv not found in PATH"))
  (let ((buf (get-buffer-create "*uv*"))
        (default-directory root))
    (with-current-buffer buf
      (let ((inhibit-read-only t)) (erase-buffer))
      (setq default-directory root))
    (message "uv %s ..." (string-join args " "))
    (make-process
     :name "uv" :buffer buf :noquery t
     :command (cons "uv" args)
     :sentinel (lambda (p _event)
                 (when (memq (process-status p) '(exit signal))
                   (if (zerop (process-exit-status p))
                       (progn (message "uv %s: done" (car args)) (funcall on-success))
                     (display-buffer buf)
                     (message "uv %s failed (see *uv*)" (car args))))))))

(defun my/venv--refresh (root)
  "Re-activate every Python buffer of project ROOT and restart its language server."
  (let (bufs)
    (dolist (b (buffer-list))
      (with-current-buffer b
        (when (and (derived-mode-p 'python-base-mode)
                   (equal (my/project-root) root))
          (my/venv-activate)
          (push b bufs))))
    (when-let* ((b (or (cl-find-if (lambda (b)
                                     (with-current-buffer b
                                       (and (fboundp 'eglot-current-server) (eglot-current-server))))
                                   bufs)
                       (car bufs))))
      (with-current-buffer b
        (when-let* ((srv (and (fboundp 'eglot-current-server) (eglot-current-server))))
          (ignore-errors (eglot-shutdown srv)))
        (eglot-ensure)))))

(defun my/venv--apply (venv root)
  "Remember VENV (nil = none) for project ROOT and apply it everywhere."
  (setf (alist-get root my/venv-choices nil nil #'equal) (or venv "none"))
  (my/venv--refresh root)
  (message "Python env: %s" (if venv (abbreviate-file-name venv) "none")))

(defun my/uv-sync (&optional root)
  "Run `uv sync' in the project, then use its .venv."
  (interactive)
  (let ((root (or root (my/project-root))))
    (my/uv--run root '("sync")
                (lambda () (my/venv--apply (expand-file-name ".venv" root) root)))))

(defun my/uv--new (root python)
  "Create ROOT/.venv with PYTHON via `uv venv', then use it."
  (let ((target (expand-file-name ".venv" root)))
    (when (or (not (file-exists-p target))
              (y-or-n-p (format "Replace existing %s? " (abbreviate-file-name target))))
      (my/uv--run root
                  (append (list "venv" "--python" python)
                          (when (file-exists-p target) '("--clear"))
                          (list target))
                  (lambda () (my/venv--apply target root))))))

(defun my/venv--dispatch (root action)
  (pcase action
    (`(use ,venv) (my/venv--apply venv root))
    (`(none)      (my/venv--apply nil root))
    (`(sync)      (my/uv-sync root))
    (`(new ,py)   (my/uv--new root py))))

(defun my/venv-select ()
  "Pick this project's Python environment with fzf."
  (interactive)
  (let* ((root  (my/project-root))
         (cands (my/venv--candidates root))
         (buf   (current-buffer))
         (act   (lambda (line)
                  (let* ((l (string-trim line))
                         (hit (or (assoc l cands)
                                  (cl-find-if (lambda (c) (string-prefix-p l (car c))) cands))))
                    (when hit
                      (with-current-buffer (if (buffer-live-p buf) buf (current-buffer))
                        (my/venv--dispatch root (cdr hit))))))))
    (if (executable-find "fzf")
        (let ((tmp (make-temp-file "venvs-")))
          (require 'fzf)
          (with-temp-file tmp (insert (mapconcat #'car cands "\n") "\n"))
          (fzf-with-command (format "cat %s" (shell-quote-argument tmp))
                            (lambda (line) (ignore-errors (delete-file tmp)) (funcall act line))))
      (funcall act (completing-read "Python env: " (mapcar #'car cands) nil t)))))

(defun my/venv-reload ()
  "Forget the remembered choice, re-detect the venv and restart the language server."
  (interactive)
  (let ((root (my/project-root)))
    (setq my/venv-choices (assoc-delete-all root my/venv-choices))
    (my/venv--refresh root)
    (message "Python env: %s" (if-let* ((v (my/venv--current))) (abbreviate-file-name v) "none"))))

  "Terminals of the current project, most recently used first."
  (let ((prefix (format "*vterm:%s/" (my/term--project-name))))
    (seq-filter (lambda (b) (string-prefix-p prefix (buffer-name b)))
                (buffer-list))))

(defun my/term-new ()
  "Open a new terminal in the project root, inheriting this buffer's environment."
  (interactive)
  (let ((default-directory (my/project-root))
        (env process-environment)
        (n 1))
    (while (get-buffer (my/term--name n)) (cl-incf n))
    (let ((process-environment env))
      (vterm (my/term--name n)))))

(defun my/term-toggle ()
  "Hide the terminal if visible, else show the last project terminal (or create one)."
  (interactive)
  (if-let* ((w (cl-find-if (lambda (w) (string-prefix-p "*vterm:" (buffer-name (window-buffer w))))
                           (window-list))))
      (if (cdr (window-list)) (delete-window w) (bury-buffer))
    (if-let* ((b (car (my/term-list))))
        (pop-to-buffer b)
      (my/term-new))))

(defun my/term--cycle (step)
  (let* ((l (sort (my/term-list)
                  (lambda (a b) (string-version-lessp (buffer-name a) (buffer-name b)))))
         (cur (cl-position (current-buffer) l)))
    (if (null l)
        (my/term-new)
      (pop-to-buffer (nth (mod (+ (or cur 0) step) (length l)) l)))))

(defun my/term-next () (interactive) (my/term--cycle 1))
(defun my/term-prev () (interactive) (my/term--cycle -1))

      (pop-to-buffer (get-buffer (completing-read "Terminal: " (mapcar #'buffer-name l) nil t)))
    (my/term-new)))
