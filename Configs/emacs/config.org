** Formatting: apheleia (async, cursor-preserving, on save)
#+begin_src emacs-lisp
(use-package apheleia
  :demand t
  :config
  (setf (alist-get 'tex-fmt apheleia-formatters) '("tex-fmt" "--stdin"))
  (dolist (m '(LaTeX-mode latex-mode))
    (setf (alist-get m apheleia-mode-alist) 'tex-fmt))
  (dolist (m '(python-mode python-ts-mode))
    (setf (alist-get m apheleia-mode-alist) '(ruff-isort ruff)))
  ;; Apheleia maps shell modes to shfmt by default; drop it if not installed.
  (unless (executable-find "shfmt")
    (dolist (m '(sh-mode bash-ts-mode))
      (setf (alist-get m apheleia-mode-alist) nil)))
  (apheleia-global-mode 1))
#+end_src

* Auto virtualenv (per Python buffer)
Finds the nearest =.venv= / =venv= above the file being edited and activates it
/buffer-locally/ (=VIRTUAL_ENV=, =PATH=, =exec-path=, =run-python=), so two
projects with different venvs can be open side by side. It runs before
=eglot-ensure=, so =ty= starts inside the right environment; =ruff= from the
venv is preferred if installed there.
#+begin_src emacs-lisp
(defvar my/venv-names '(".venv" "PyEnv"))

(defun my/venv-find (&optional dir)
  "Nearest virtualenv directory above DIR (default: current buffer's dir)."
  (let ((dir (or dir default-directory)))
    (cl-some (lambda (name)
               (when-let* ((root (locate-dominating-file dir name))
                           (venv (expand-file-name name root)))
                 (and (file-executable-p (expand-file-name "bin/python" venv)) venv)))
             my/venv-names)))

(defun my/venv-activate ()
  "Buffer-locally activate the virtualenv belonging to this Python buffer."
  (when-let* ((venv (my/venv-find)))
    (let ((bin  (expand-file-name "bin" venv))
          (path (getenv-internal "PATH" (default-value 'process-environment))))
      (setq-local python-shell-virtualenv-root venv)
      (setq-local exec-path (cons bin (remove bin (default-value 'exec-path))))
      (setq-local process-environment
                  (append (list (concat "VIRTUAL_ENV=" venv)
                                (concat "PATH=" bin ":" path))
                          (cl-remove-if
                           (lambda (e) (string-match-p "\\`\\(VIRTUAL_ENV\\|PATH\\)=" e))
                           (default-value 'process-environment)))))))

(defun my/venv-reload ()
  "Re-detect the venv for this buffer and restart its language server."
  (interactive)
  (my/venv-activate)
  (when-let* ((srv (and (fboundp 'eglot-current-server) (eglot-current-server))))
    (eglot-reconnect srv))
  (message "venv: %s" (or python-shell-virtualenv-root "none found")))

;; depth -90: run before `eglot-ensure' so the server inherits the env
(add-hook 'python-base-mode-hook #'my/venv-activate -90)
#+end_src

Needs =cmake=, =libtool= and =libvterm= headers (Emacs offers to compile the
module on first use). Colors follow the pywal =term-color-*= faces.
Esc leaves insert state in a terminal, so =SPC= (leader) and =C-h/j/k/l= work
there; =i= goes back to typing in the shell. =C-`= toggles from any state.
#+begin_src emacs-lisp
(use-package vterm
  :ensure t
  :custom
  (vterm-max-scrollback 20000)
  (vterm-kill-buffer-on-exit t)
  (vterm-timer-delay 0.01)
  :config
  (evil-set-initial-state 'vterm-mode 'insert))

;; Terminals open in a bottom side window (one-third of the frame).
(add-to-list 'display-buffer-alist
             '("\\`\\*vterm:"
               (display-buffer-in-side-window)
               (side . bottom)
               (slot . 0)
               (window-height . 0.33)
               (window-parameters . ((no-delete-other-windows . t)))))

(defun my/term--project-name ()
  (if-let* ((p (project-current)))
      (file-name-nondirectory (directory-file-name (project-root p)))
    "global"))

(defun my/term--name (n)
  (format "*vterm:%s/%d*" (my/term--project-name) n))

(defun my/term-list ()
  "Terminals of the current project, most recently used first."
  (let ((prefix (format "*vterm:%s/" (my/term--project-name))))
    (seq-filter (lambda (b) (string-prefix-p prefix (buffer-name b)))
                (buffer-list))))

(defun my/term-new ()
  "Open a new terminal in the project root."
  (interactive)
  (let ((default-directory (my/project-root))
        (n 1))
    (while (get-buffer (my/term--name n)) (cl-incf n))
    (vterm (my/term--name n))))

(defun my/term-toggle ()
  "Hide the terminal if visible, else show the last project terminal (or create one)."
  (interactive)
  (if-let* ((w (cl-find-if (lambda (w) (string-prefix-p "*vterm:" (buffer-name (window-buffer w))))
                           (window-list))))
      (if (cdr (window-list)) (delete-window w) (bury-buffer))
    (if-let* ((b (car (my/term-list))))
        (pop-to-buffer b)
      (my/term-new))))

(defun my/term--cycle (step)
  (let* ((l (sort (my/term-list)
                  (lambda (a b) (string-version-lessp (buffer-name a) (buffer-name b)))))
         (cur (cl-position (current-buffer) l)))
    (if (null l)
        (my/term-new)
      (pop-to-buffer (nth (mod (+ (or cur 0) step) (length l)) l)))))

(defun my/term-next () (interactive) (my/term--cycle 1))
(defun my/term-prev () (interactive) (my/term--cycle -1))

(defun my/term-switch ()
  "Pick one of this project's terminals."
  (interactive)
  (if-let* ((l (my/term-list)))
      (pop-to-buffer (get-buffer (completing-read "Terminal: " (mapcar #'buffer-name l) nil t)))
    (my/term-new)))

(defun my/term-kill ()
  "Kill the current terminal (and its window)."
  (interactive)
  (when (derived-mode-p 'vterm-mode)
    (let ((kill-buffer-query-functions nil))
      (kill-buffer (current-buffer)))))
